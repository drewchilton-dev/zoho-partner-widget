function sendChatMessage() { var inputEl = document.getElementById("userMessage"); var btnEl = document.getElementById("sendBtn"); var messageText = inputEl.value.trim(); if (!messageText) return; var name = currentUser ? (currentUser.full_name || currentUser.first_name) : "CRM User"; var email = currentUser ? currentUser.email : "Unknown Email"; inputEl.disabled = true; btnEl.disabled = true; inputEl.value = ""; inputEl.placeholder = "Sending..."; // Format required by Zoho CRM JS SDK var funcData = { arguments: JSON.stringify({ "user_name": name, "user_email": email, "user_message": messageText }) }; ZOHO.CRM.FUNCTIONS.execute("send_home_page_chat", funcData).then(function(data) { inputEl.placeholder = "Sent to Operations! ✓"; setTimeout(function() { inputEl.placeholder = "Chat Operations..."; inputEl.disabled = false; btnEl.disabled = false; hideChatInput(); }, 2000); }).catch(function(err) { console.error("Zoho function execution error:", err); inputEl.placeholder = "Error sending!"; setTimeout(function() { inputEl.placeholder = "Chat Operations..."; inputEl.disabled = false; btnEl.disabled = false; }, 2000); }); }
